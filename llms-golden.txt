// golden: exit status from main logic
// idiom: compute an exit code in a pure function; main reports it and only then calls process::exit
use std::process

fn check(args: Vec<string>) -> int {
    if args.len() == 0 {
        println("error: no input given")
        return 2
    }
    for a in args {
        if a.parse::<int>().is_err() {
            println("error: ${a} is not a number")
            return 1
        }
    }
    println("all ${args.len()} inputs are numbers")
    0
}

fn label(code: int) -> string {
    match code {
        0 => "success",
        1 => "bad input",
        2 => "usage error",
        _ => "unknown",
    }
}

fn main() {
    for args in [["1", "2", "3"], ["4", "x"], []] {
        let code = check(args)
        println("exit ${code} (${label(code)})")
    }
    let final_code = check(["7"])
    if final_code != 0 {
        process.exit(final_code)
    }
    println("finished cleanly")
}

// golden: command-line flag parsing
// idiom: a struct of options with defaults; scan the argument Vec with an index; Result reports bad input
struct Options {
    name: string,
    count: int,
    verbose: bool,
    files: Vec<string>,
}

fn parse(args: Vec<string>) -> Result<Options, string> {
    let mut opts = Options { name: "world", count: 1, verbose: false, files: [] }
    let mut i = 0
    while i < args.len() {
        let arg = args[i]
        if arg == "--verbose" || arg == "-v" {
            opts.verbose = true
        } else if arg == "--name" || arg == "-n" {
            i += 1
            if i >= args.len() {
                return Err("${arg} needs a value")
            }
            opts.name = args[i]
        } else if arg == "--count" || arg == "-c" {
            i += 1
            if i >= args.len() {
                return Err("${arg} needs a value")
            }
            opts.count = args[i].parse::<int>().map_err(|_| "bad count: ${args[i]}")?
        } else if arg.starts_with("-") {
            return Err("unknown flag: ${arg}")
        } else {
            opts.files.push(arg)
        }
        i += 1
    }
    Ok(opts)
}

fn run(args: Vec<string>) {
    match parse(args) {
        Ok(o) => {
            for _ in 0..o.count {
                println("hello, ${o.name}")
            }
            println("verbose=${o.verbose} files=${o.files:?}")
        },
        Err(e) => println("error: ${e}"),
    }
}

fn main() {
    run(["--name", "Ada", "-c", "2", "-v", "a.txt", "b.txt"])
    run([])
    run(["--count", "many"])
    run(["--name"])
    run(["--bogus"])
}

// golden: key=value arguments
// idiom: split each argument once on "="; a HashMap holds the settings; get() returns an Option
use std::collections::HashMap

fn parse_settings(args: Vec<string>) -> HashMap<string, string> {
    let mut settings: HashMap<string, string> = HashMap::new()
    for arg in args {
        if arg.contains("=") {
            let pieces: Vec<string> = arg.split("=").collect()
            settings.insert(pieces[0], pieces[1])
        }
    }
    settings
}

fn lookup(settings: HashMap<string, string>, key: string, fallback: string) -> string {
    settings.get(key) ?? fallback
}

fn main() {
    let args = ["port=9000", "debug=true", "ignored", "name=svc"]
    let settings = parse_settings(args)
    println("port: ${lookup(settings, "port", "8080")}")
    println("host: ${lookup(settings, "host", "localhost")}")
    println("name: ${lookup(settings, "name", "unnamed")}")
    let debug = lookup(settings, "debug", "false") == "true"
    println("debug: ${debug}")
    let mut keys: Vec<string> = settings.keys().collect()
    keys.sort()
    println("keys: ${keys.join(", ")}")
    println("count: ${settings.len()}")
}

// golden: subcommand dispatch
// idiom: match on the first argument string; each command is a function; usage text for the fallback
fn usage() -> string {
    "usage: tool <add|mul|neg|help> [numbers...]"
}

fn numbers(args: Vec<string>) -> Result<Vec<int>, string> {
    let mut out: Vec<int> = []
    for a in args {
        out.push(a.parse::<int>().map_err(|_| "not a number: ${a}")?)
    }
    Ok(out)
}

fn add(nums: Vec<int>) -> int {
    let mut total = 0
    for n in nums {
        total += n
    }
    total
}

fn mul(nums: Vec<int>) -> int {
    let mut total = 1
    for n in nums {
        total *= n
    }
    total
}

fn dispatch(args: Vec<string>) -> string {
    if args.len() == 0 {
        return usage()
    }
    let rest: Vec<string> = args.skip(1).collect()
    match args[0] {
        "add" => match numbers(rest) {
            Ok(n) => "sum = ${add(n)}",
            Err(e) => "error: ${e}",
        },
        "mul" => match numbers(rest) {
            Ok(n) => "product = ${mul(n)}",
            Err(e) => "error: ${e}",
        },
        "neg" => match numbers(rest) {
            Ok(n) => "negated = ${n.map(|x| 0 - x).collect():?}",
            Err(e) => "error: ${e}",
        },
        "help" => usage(),
        other => "unknown command '${other}'; ${usage()}",
    }
}

fn main() {
    println(dispatch(["add", "1", "2", "3"]))
    println(dispatch(["mul", "4", "5"]))
    println(dispatch(["neg", "1", "-2"]))
    println(dispatch(["add", "1", "x"]))
    println(dispatch(["frobnicate"]))
    println(dispatch([]))
}

// golden: help text from a table
// idiom: a Vec of structs describes the flags once; help and parsing both read it; padding aligns the columns
use std::strings

struct Flag {
    short: string,
    long: string,
    help: string,
}

fn flags() -> Vec<Flag> {
    [
        Flag { short: "-h", long: "--help", help: "show this message" },
        Flag { short: "-v", long: "--verbose", help: "print more detail" },
        Flag { short: "-o", long: "--output", help: "write results to a file" },
        Flag { short: "-q", long: "--quiet", help: "print nothing but errors" },
    ]
}

fn pad(text: string, width: int) -> string {
    let missing = width - text.len() as int
    if missing <= 0 { text } else { "${text}${strings.repeat(" ", missing)}" }
}

fn help() -> string {
    let mut longest: int = 0
    for f in flags() {
        let n: int = f.long.len() as int
        if n > longest {
            longest = n
        }
    }
    let mut out = "options:"
    for f in flags() {
        out += "\n  ${f.short}, ${pad(f.long, longest)}  ${f.help}"
    }
    out
}

fn is_known(arg: string) -> bool {
    for f in flags() {
        if arg == f.short || arg == f.long {
            return true
        }
    }
    false
}

fn main() {
    println(help())
    for arg in ["-v", "--output", "--nope"] {
        println("${arg} known: ${is_known(arg)}")
    }
}
--- output ---
options:
  -h, --help     show this message
  -v, --verbose  print more detail
  -o, --output   write results to a file
  -q, --quiet    print nothing but errors
-v known: true
--output known: true
--nope known: false

// golden: binary search
// idiom: integer midpoint arithmetic on a sorted Vec; return Option<int> for "maybe found"
fn find(items: Vec<int>, target: int) -> Option<int> {
    if items.len() == 0 {
        return None
    }
    let mut low: int = 0
    let mut high: int = items.len() as int - 1
    while low <= high {
        let mid = low + (high - low) / 2
        if items[mid] == target {
            return Some(mid)
        }
        if items[mid] < target {
            low = mid + 1
        } else {
            if mid == 0 {
                return None
            }
            high = mid - 1
        }
    }
    None
}

fn describe(items: Vec<int>, target: int) -> string {
    match find(items, target) {
        Some(i) => "${target} is at index ${i}",
        None => "${target} is not present",
    }
}

fn main() {
    let items = [2, 3, 5, 7, 11, 13, 17, 19, 23]
    for target in [7, 2, 23, 4, 30] {
        println(describe(items, target))
    }
    let empty: Vec<int> = []
    println(describe(empty, 1))
}
--- output ---

2 is at index 0
23 is at index 8
4 is not present

// golden: group records by key
// idiom: HashMap<K, Vec<V>> with entry().or_insert([]).push(); sort keys for deterministic output
use std::collections::HashMap

struct Employee {
    name: string,
    dept: string,
    salary: int,
}

fn by_dept(staff: Vec<Employee>) -> HashMap<string, Vec<Employee>> {
    let mut groups: HashMap<string, Vec<Employee>> = HashMap::new()
    for e in staff {
        groups.entry(e.dept).or_insert([]).push(e)
    }
    groups
}

fn payroll(staff: Vec<Employee>) -> int {
    let mut sum = 0
    for e in staff {
        sum += e.salary
    }
    sum
}

fn main() {
    let staff = [
        Employee { name: "Ada", dept: "eng", salary: 120 },
        Employee { name: "Linus", dept: "eng", salary: 110 },
        Employee { name: "Grace", dept: "ops", salary: 100 },
        Employee { name: "Ken", dept: "ops", salary: 95 },
        Employee { name: "Barbara", dept: "research", salary: 130 },
    ]
    let groups = by_dept(staff)
    let mut depts: Vec<string> = groups.keys().collect()
    depts.sort()
    for dept in depts {
        let members = groups[dept]
        let names: Vec<string> = members.map(|e| e.name).collect()
        println("${dept}: ${names.join(", ")} (payroll ${payroll(members)})")
    }
}

ops: Grace, Ken (payroll 195)

// golden: matrix multiply and transpose
// idiom: Vec<Vec<int>> indexed with m[i][j]; nested for ranges; build rows with push
fn multiply(a: Vec<Vec<int>>, b: Vec<Vec<int>>) -> Vec<Vec<int>> {
    let mut out: Vec<Vec<int>> = []
    for i in 0..a.len() {
        let mut row: Vec<int> = []
        for j in 0..b[0].len() {
            let mut sum = 0
            for k in 0..b.len() {
                sum += a[i][k] * b[k][j]
            }
            row.push(sum)
        }
        out.push(row)
    }
    out
}

fn transpose(m: Vec<Vec<int>>) -> Vec<Vec<int>> {
    let mut out: Vec<Vec<int>> = []
    for j in 0..m[0].len() {
        let mut row: Vec<int> = []
        for i in 0..m.len() {
            row.push(m[i][j])
        }
        out.push(row)
    }
    out
}

fn show(m: Vec<Vec<int>>) {
    for row in m {
        let cells: Vec<string> = row.map(|x| "${x:>4}").collect()
        println(cells.join(""))
    }
}

fn main() {
    let a = [[1, 2, 3], [4, 5, 6]]
    let b = [[7, 8], [9, 10], [11, 12]]
    println("a x b:")
    show(multiply(a, b))
    println("transpose of a:")
    show(transpose(a))
}
--- output ---

  58  64
 139 154
transpose of a:
   1   4
   2   5
   3   6

// golden: set operations
// idiom: HashSet for membership and uniqueness; collect a sorted Vec before printing
use std::collections::HashSet

fn to_set(items: Vec<int>) -> HashSet<int> {
    let mut set: HashSet<int> = HashSet::new()
    for x in items {
        set.insert(x)
    }
    set
}

fn sorted(set: HashSet<int>) -> Vec<int> {
    let mut items: Vec<int> = []
    for x in set {
        items.push(x)
    }
    items.sort()
    items
}

fn union(a: HashSet<int>, b: HashSet<int>) -> HashSet<int> {
    let mut out = a
    for x in b {
        out.insert(x)
    }
    out
}

fn intersection(a: HashSet<int>, b: HashSet<int>) -> HashSet<int> {
    let mut out: HashSet<int> = HashSet::new()
    for x in a {
        if b.contains(x) {
            out.insert(x)
        }
    }
    out
}

fn main() {
    let a = to_set([1, 2, 3, 4, 4, 5])
    let b = to_set([4, 5, 6, 7])
    println("a = ${sorted(a):?}")
    println("b = ${sorted(b):?}")
    println("union = ${sorted(union(a, b)):?}")
    println("intersection = ${sorted(intersection(a, b)):?}")
    println("a has 3: ${a.contains(3)}, b has 3: ${b.contains(3)}")
    println("unique count: ${a.len()}")
}

// golden: sort records by several keys
// idiom: sort_by with a comparison closure; cmp() and then() chain keys, b.cmp(a) reverses
struct Player {
    name: string,
    score: int,
    age: int,
}

fn show(players: Vec<Player>) {
    for p in players {
        println("${p.name} score=${p.score} age=${p.age}")
    }
}

fn main() {
    let mut players = [
        Player { name: "Mia", score: 90, age: 31 },
        Player { name: "Noah", score: 85, age: 24 },
        Player { name: "Zoe", score: 90, age: 22 },
        Player { name: "Eli", score: 85, age: 24 },
    ]
    println("by score descending, then name:")
    players.sort_by(|a, b| b.score.cmp(a.score).then(a.name.cmp(b.name)))

    println("by age ascending, then name:")
    players.sort_by(|a, b| a.age.cmp(b.age).then(a.name.cmp(b.name)))
    show(players)
    let mut scores: Vec<int> = players.map(|p| p.score).collect()
    scores.sort()
    scores.reverse()
    println("scores ${scores:?}")
}

// golden: stack and queue
// idiom: Vec as a stack with push/pop (pop returns an Option); VecDeque for a queue
use std::collections::VecDeque

fn balanced(text: string) -> bool {
    let mut stack: Vec<char> = []
    for c in text.chars() {
        if c == '(' || c == '[' || c == '{' {
            stack.push(c)
        } else if c == ')' || c == ']' || c == '}' {
            match stack.pop() {
                Some(open) => {
                    let ok = (open == '(' && c == ')') || (open == '[' && c == ']') || (open == '{' && c == '}')
                    if !ok {
                        return false
                    }
                },
                None => return false,
            }
        }
    }
    stack.len() == 0
}

fn main() {
    for text in ["(a[b]{c})", "(]", "((", ""] {
        println("\"${text}\" balanced: ${balanced(text)}")
    }
    let mut line: VecDeque<string> = VecDeque::new()
    line.push_back("ann")
    line.push_back("bob")
    line.push_back("cy")
    println("serving ${line.pop_front() ?? "nobody"}")
    line.push_back("di")
    println("next ${line.pop_front() ?? "nobody"}, waiting ${line.len()}")
    let mut stack = [1, 2, 3]
    stack.push(4)
    println("top ${stack.pop() ?? 0}, size ${stack.len()}")
}
--- output ---

"(]" balanced: false
"((" balanced: false
"" balanced: true

=== collections_top_k.wj ===
// golden: top-k and dedupe
// idiom: copy into a new Vec before sorting; sort then reverse for descending; plain loops for max, min and the average
fn top_k(items: Vec<int>, k: int) -> Vec<int> {
    let mut sorted: Vec<int> = []
    for x in items {
        sorted.push(x)
    }
    sorted.sort()
    sorted.reverse()
    let mut out: Vec<int> = []
    for i in 0..k {
        out.push(sorted[i])
    }
    out
}

fn dedupe(items: Vec<int>) -> Vec<int> {
    let mut sorted = items
    sorted.sort()
    let mut out: Vec<int> = []
    for x in sorted {
        if out.len() == 0 || out[out.len() - 1] != x {
            out.push(x)
        }
    }
    out
}

fn average(items: Vec<int>) -> float {
    if items.len() == 0 {
        return 0.0
    }
    let mut sum = 0
    for x in items {
        sum += x
    }
    sum as float / items.len() as float
}

fn largest(items: Vec<int>) -> int {
    let mut best = items[0]
    for x in items {
        if x > best {
            best = x
        }
    }
    best
}

fn smallest(items: Vec<int>) -> int {
    let mut best = items[0]
    for x in items {
        if x < best {
            best = x
        }
    }
    best
}

fn main() {
    let scores = [72, 91, 85, 91, 60, 85, 99, 72]
    println("top 3: ${top_k(scores, 3):?}")
    println("distinct: ${dedupe(scores):?}")
    println("average: ${average(scores):.2}")
    println("max ${largest(scores)} min ${smallest(scores)}")
}
--- output ---
top 3: [99, 91, 91]
distinct: [60, 72, 85, 91, 99]
average: 81.88
max 99 min 60

=== errors_custom_enum.wj ===
// golden: custom error enum
// idiom: an enum names every failure; a message method turns it into text; match handles each case
enum BankError {
    Insufficient(int, int),
    UnknownAccount(string),
    Frozen,
}

fn message(e: BankError) -> string {
    match e {
        BankError::Insufficient(have, need) => "insufficient funds: have ${have}, need ${need}",
        BankError::UnknownAccount(name) => "no such account: ${name}",
        BankError::Frozen => "account is frozen",
    }
}

struct Account {
    name: string,
    balance: int,
    frozen: bool,
}

fn withdraw(acct: Account, amount: int) -> Result<Account, BankError> {
    if acct.frozen {
        return Err(BankError::Frozen)
    }
    if acct.balance < amount {
        return Err(BankError::Insufficient(acct.balance, amount))
    }
    Ok(Account { name: acct.name, balance: acct.balance - amount, frozen: false })
}

fn lookup(accounts: Vec<Account>, name: string) -> Result<Account, BankError> {
    for a in accounts {
        if a.name == name {
            return Ok(a)
        }
    }
    Err(BankError::UnknownAccount(name))
}

fn main() {
    let accounts = [
        Account { name: "ann", balance: 100, frozen: false },
        Account { name: "bob", balance: 20, frozen: false },
        Account { name: "cy", balance: 500, frozen: true },
    ]
    for (name, amount) in [("ann", 30), ("bob", 50), ("cy", 10), ("dee", 5)] {
        let result = lookup(accounts, name).and_then(|a| withdraw(a, amount))
        match result {
            Ok(a) => println("${name}: ok, balance now ${a.balance}"),
            Err(e) => println("${name}: ${message(e)}"),
        }
    }
}
--- output ---

bob: insufficient funds: have 20, need 50
cy: account is frozen

// golden: Option to Result with ok_or
// idiom: ok_or turns a missing value into an error; chain lookups with `?` inside a function returning Result
use std::collections::HashMap

struct Config {
    values: HashMap<string, string>,
}

impl Config {
    fn get(self, key: string) -> Result<string, string> {
        self.values.get(key).ok_or("missing key: ${key}")
    }

    fn get_int(self, key: string) -> Result<int, string> {
        let text = self.get(key)?
        text.parse::<int>().map_err(|_| "key ${key} is not a number: ${text}")
    }
}

fn describe(cfg: Config) -> Result<string, string> {
    let host = cfg.get("host")?
    let port = cfg.get_int("port")?
    let workers = cfg.get_int("workers")?
    Ok("${host}:${port} with ${workers} workers")
}

fn main() {
    let mut values: HashMap<string, string> = HashMap::new()
    values.insert("host", "localhost")
    values.insert("port", "8080")
    let partial = Config { values: values }
    println("${describe(partial):?}")
    let mut more: HashMap<string, string> = HashMap::new()
    more.insert("host", "example.org")
    more.insert("port", "80")
    more.insert("workers", "4")
    println("${describe(Config { values: more }):?}")
    let mut bad: HashMap<string, string> = HashMap::new()
    bad.insert("host", "h")
    bad.insert("port", "eighty")
    println("${describe(Config { values: bad }):?}")
}

// golden: Result and the ? operator
// idiom: return Result<T, string>; `?` returns early on Err; map_err gives parse errors a readable message
fn parse_int(text: string) -> Result<int, string> {
    text.parse::<int>().map_err(|_| "not a number: ${text}")
}

fn parse_pair(text: string) -> Result<(int, int), string> {
    let parts: Vec<string> = text.split(",").collect()
    if parts.len() != 2 {
        return Err("expected two numbers separated by a comma: ${text}")
    }
    let a = parse_int(parts[0])?
    let b = parse_int(parts[1])?
    Ok((a, b))
}

fn divide(text: string) -> Result<int, string> {
    let pair = parse_pair(text)?
    if pair.1 == 0 {
        return Err("division by zero in ${text}")
    }
    Ok(pair.0 / pair.1)
}

fn main() {
    let inputs = ["84,2", "7,0", "x,3", "1;2", "100,7"]
    for input in inputs {
        match divide(input) {
            Ok(n) => println("${input} -> ${n}"),
            Err(e) => println("${input} -> error: ${e}"),
        }
    }
    let total = divide("10,2").unwrap_or(0) + divide("oops").unwrap_or(-1)
    println("total ${total}")
}
--- output ---

x,3 -> error: not a number: x
1;2 -> error: expected two numbers separated by a comma: 1;2

// golden: retry with a counter
// idiom: loop with an attempt counter; a Result-returning step is retried until Ok or the budget runs out
fn flaky(attempt: int) -> Result<string, string> {
    if attempt < 3 {
        Err("timeout on attempt ${attempt}")
    } else {
        Ok("connected on attempt ${attempt}")
    }
}

fn with_retries(max_attempts: int) -> Result<string, string> {
    let mut attempt = 1
    let mut last_error = "never tried"
    while attempt <= max_attempts {
        match flaky(attempt) {
            Ok(msg) => return Ok(msg),
            Err(e) => {
                println("  retrying after: ${e}")
                last_error = e
            },
        }
        attempt += 1
    }
    Err("gave up after ${max_attempts} attempts, last error: ${last_error}")
}

fn backoff_ms(attempt: int) -> int {
    let mut delay = 100
    for _ in 1..attempt {
        delay *= 2
    }
    delay
}

fn main() {
    println("budget 5:")
    println("${with_retries(5):?}")
    println("budget 2:")
    println("${with_retries(2):?}")
    for n in 1..=5 {
        println("attempt ${n} waits ${backoff_ms(n)}ms")
    }
}

// golden: breadth-first search on a grid
// idiom: VecDeque as the frontier; a HashMap of distances doubles as the visited set; neighbors from a direction list
use std::collections::VecDeque
use std::collections::HashMap

fn solve(maze: Vec<string>) -> Option<int> {
    let rows: int = maze.len() as int
    let cols: int = maze[0].len() as int
    let mut dist: HashMap<int, int> = HashMap::new()
    let mut queue: VecDeque<(int, int)> = VecDeque::new()
    queue.push_back((0, 0))
    dist.insert(0, 0)
    let steps = [(1, 0), (-1, 0), (0, 1), (0, -1)]
    while let Some((r, c)) = queue.pop_front() {
        let here = dist[r * cols + c]
        if r == rows - 1 && c == cols - 1 {
            return Some(here)
        }
        for (dr, dc) in steps {
            let nr = r + dr
            let nc = c + dc
            if nr < 0 || nr >= rows || nc < 0 || nc >= cols {
                continue
            }
            let wall = maze[nr][nc..nc + 1] == "#"
            if !wall && !dist.contains_key(nr * cols + nc) {
                dist.insert(nr * cols + nc, here + 1)
                queue.push_back((nr, nc))
            }
        }
    }
    None
}

fn main() {
    let open = ["..#.", "..#.", "....", "#..."]
    let blocked = ["..#.", "###.", "....", "...."]
    match solve(open) {
        Some(n) => println("open maze: ${n} steps"),
        None => println("open maze: no path"),
    }
    match solve(blocked) {
        Some(n) => println("blocked maze: ${n} steps"),
        None => println("blocked maze: no path"),
    }
}

// golden: connected components with depth-first search
// idiom: adjacency list as Vec<Vec<int>>; an explicit stack instead of recursion; a visited Vec<bool>
fn build(n: int, edges: Vec<(int, int)>) -> Vec<Vec<int>> {
    let mut adj: Vec<Vec<int>> = []
    for _ in 0..n {
        adj.push([])
    }
    for (a, b) in edges {
        adj[a].push(b)
        adj[b].push(a)
    }
    adj
}

fn components(adj: Vec<Vec<int>>) -> Vec<Vec<int>> {
    let mut seen: Vec<bool> = []
    for _ in 0..adj.len() {
        seen.push(false)
    }
    let mut groups: Vec<Vec<int>> = []
    for start in 0..adj.len() {
        if seen[start] {
            continue
        }
        let mut group: Vec<int> = []
        let mut stack = [start]
        seen[start] = true
        while stack.len() > 0 {
            let node = stack.pop() ?? 0
            group.push(node)
            for next in adj[node] {
                if !seen[next] {
                    seen[next] = true
                    stack.push(next)
                }
            }
        }
        group.sort()
        groups.push(group)
    }
    groups
}

fn main() {
    let adj = build(8, [(0, 1), (1, 2), (3, 4), (5, 6), (6, 7), (7, 5)])
    let groups = components(adj)
    println("${groups.len()} components")
    for g in groups {
        println("  ${g:?}")
    }
}

// golden: shortest paths with Dijkstra
// idiom: edges as (to, weight) tuples per node; a simple "pick the closest unvisited node" loop, no priority queue needed at this size
fn dijkstra(graph: Vec<Vec<(int, int)>>, start: int) -> Vec<int> {
    let infinity = 1000000000
    let mut dist: Vec<int> = []
    let mut done: Vec<bool> = []
    for _ in 0..graph.len() {
        dist.push(infinity)
        done.push(false)
    }
    dist[start] = 0
    for _ in 0..graph.len() {
        let mut best = -1
        for i in 0..graph.len() {
            if !done[i] && dist[i] < infinity && (best == -1 || dist[i] < dist[best]) {
                best = i
            }
        }
        if best == -1 {
            break
        }
        done[best] = true
        for (to, weight) in graph[best] {
            if dist[best] + weight < dist[to] {
                dist[to] = dist[best] + weight
            }
        }
    }
    dist
}

fn main() {
    let graph = [
        [(1, 7), (2, 9), (5, 14)],
        [(0, 7), (2, 10), (3, 15)],
        [(0, 9), (1, 10), (3, 11), (5, 2)],
        [(1, 15), (2, 11), (4, 6)],
        [(3, 6), (5, 9)],
        [(0, 14), (2, 2), (4, 9)],
        [],
    ]
    let dist = dijkstra(graph, 0)
    for i in 0..dist.len() {
        if dist[i] == 1000000000 {
            println("node ${i}: unreachable")
        } else {
            println("node ${i}: ${dist[i]}")
        }
    }
}

// golden: degrees of separation
// idiom: HashMap<string, Vec<string>> as the social graph; breadth-first levels using two Vecs (current and next)
use std::collections::HashMap
use std::collections::HashSet

fn friends_of(graph: HashMap<string, Vec<string>>, name: string) -> Vec<string> {
    graph.get(name) ?? []
}

fn distance(graph: HashMap<string, Vec<string>>, from: string, to: string) -> Option<int> {
    if from == to {
        return Some(0)
    }
    let mut seen: HashSet<string> = HashSet::new()
    seen.insert(from)
    let mut current = [from]
    let mut level = 0
    while current.len() > 0 {
        level += 1
        let mut next: Vec<string> = []
        for person in current {
            for friend in friends_of(graph, person) {
                if friend == to {
                    return Some(level)
                }
                if !seen.contains(friend) {
                    seen.insert(friend)
                    next.push(friend)
                }
            }
        }
        current = next
    }
    None
}

fn main() {
    let mut graph: HashMap<string, Vec<string>> = HashMap::new()
    let links = [("ann", "bob"), ("bob", "cy"), ("cy", "di"), ("ann", "eve"), ("zed", "yan")]
    for (a, b) in links {
        graph.entry(a).or_insert([]).push(b)
        graph.entry(b).or_insert([]).push(a)
    }
    for (a, b) in [("ann", "di"), ("ann", "ann"), ("eve", "cy"), ("ann", "zed")] {
        match distance(graph, a, b) {

// golden: topological sort of tasks
// idiom: count incoming edges (Kahn's algorithm); a Vec as the ready list; report a cycle when not every node is placed
use std::collections::HashMap

fn order(tasks: Vec<string>, needs: Vec<(string, string)>) -> Result<Vec<string>, string> {
    let mut incoming: HashMap<string, int> = HashMap::new()
    let mut after: HashMap<string, Vec<string>> = HashMap::new()
    for t in tasks {
        incoming.insert(t, 0)
        after.insert(t, [])
    }
    for (first, then) in needs {
        incoming.insert(then, incoming[then] + 1)
        after.entry(first).or_insert([]).push(then)
    }
    let mut ready: Vec<string> = []
    for t in tasks {
        if incoming[t] == 0 {
            ready.push(t)
        }
    }
    let mut result: Vec<string> = []
    while ready.len() > 0 {
        ready.sort()
        let next = ready.remove(0)
        result.push(next)
        for follower in after[next] {
            incoming.insert(follower, incoming[follower] - 1)
            if incoming[follower] == 0 {
                ready.push(follower)
            }
        }
    }
    if result.len() != tasks.len() {
        return Err("dependency cycle detected")
    }
    Ok(result)
}

fn main() {
    let tasks = ["compile", "link", "test", "package", "lint"]
    let needs = [("compile", "link"), ("link", "test"), ("link", "package"), ("lint", "test")]
    match order(tasks, needs) {
        Ok(steps) => println("order: ${steps.join(" -> ")}"),
        Err(e) => println("error: ${e}"),
    }
    let cyclic = [("compile", "link"), ("link", "compile")]
    match order(["compile", "link"], cyclic) {
        Ok(steps) => println("order: ${steps.join(" -> ")}"),
        Err(e) => println("error: ${e}"),
    }
}

// golden: URL parts and percent-decoding
// idiom: split a URL into scheme, host and path with plain string methods; decode %XX by hand over chars
fn hex_value(c: char) -> int {
    if c >= '0' && c <= '9' {
        return c as int - '0' as int
    }
    if c >= 'a' && c <= 'f' {
        return c as int - 'a' as int + 10
    }
    if c >= 'A' && c <= 'F' {
        return c as int - 'A' as int + 10
    }
    -1
}

fn decode(text: string) -> string {
    let cs: Vec<char> = text.chars().collect()
    let mut out = ""
    let mut i = 0
    while i < cs.len() {
        if cs[i] == '%' && i + 2 < cs.len() + 0 && hex_value(cs[i + 1]) >= 0 && hex_value(cs[i + 2]) >= 0 {
            let code = hex_value(cs[i + 1]) * 16 + hex_value(cs[i + 2])
            out += "${(code as u8) as char}"
            i += 3
        } else if cs[i] == '+' {
            out = "${out} "
            i += 1
        } else {
            out += "${cs[i]}"
            i += 1
        }
    }
    out
}

fn host_of(url: string) -> string {
    let after: Vec<string> = url.split("://").collect()
    if after.len() < 2 {
        return ""
    }
    let rest: Vec<string> = after[1].split("/").collect()
    rest[0]
}

// golden: parse a raw HTTP request
// idiom: split the text into lines, then the request line into parts; a struct holds the result and Result reports malformed input
use std::collections::HashMap

struct Request {
    method: string,
    path: string,
    query: HashMap<string, string>,
    headers: HashMap<string, string>,
}

fn parse_query(text: string) -> HashMap<string, string> {
    let mut query: HashMap<string, string> = HashMap::new()
    for pair in text.split("&") {
        let kv: Vec<string> = pair.split("=").collect()
        if kv.len() == 2 {
            query.insert(kv[0], kv[1])
        }
    }
    query
}

fn parse(raw: string) -> Result<Request, string> {
    let lines: Vec<string> = raw.lines().collect()
    if lines.len() == 0 {
        return Err("empty request")
    }
    let first: Vec<string> = lines[0].split(" ").collect()
    if first.len() != 3 {
        return Err("bad request line: ${lines[0]}")
    }
    let target: Vec<string> = first[1].split("?").collect()
    let query = if target.len() == 2 { parse_query(target[1]) } else { HashMap::new() }
    let mut headers: HashMap<string, string> = HashMap::new()
    for line in lines.skip(1) {
        if line.contains(": ") {
            let hv: Vec<string> = line.split(": ").collect()
            headers.insert(hv[0].to_lowercase(), hv[1])
        }
    }
    Ok(Request { method: first[0], path: target[0], query: query, headers: headers })
}

fn main() {
    let raw = "GET /search?q=wind&page=2 HTTP/1.1\nHost: example.org\nAccept: text/html\nX-Trace: abc"
    match parse(raw) {

            println("q = ${r.query["q"]}, page = ${r.query["page"]}")
            println("host = ${r.headers["host"]}")
            println("${r.headers.len()} headers")
        },
        Err(e) => println("error: ${e}"),
    }
    println("${parse("nonsense").is_err()}")

}

// golden: build an HTTP response
// idiom: join header lines with a blank line before the body; one function per concern, all pure
use std::strings

struct Response {
    status: int,
    headers: Vec<(string, string)>,
    body: string,
}

fn reason(status: int) -> string {
    match status {
        200 => "OK",
        201 => "Created",
        404 => "Not Found",
        500 => "Internal Server Error",
        _ => "Unknown",
    }
}

fn with_header(r: Response, name: string, value: string) -> Response {
    let mut headers = r.headers
    headers.push((name, value))
    Response { status: r.status, headers: headers, body: r.body }
}

}

fn not_found(path: string) -> Response {

}

fn user_page(id: string) -> Response {
    match id.parse::<int>() {
        Ok(n) => ok("user #${n}"),
        Err(_) => Response { status: 400, body: "bad user id: ${id}" },

    let parts: Vec<string> = path.split("/").collect()
    match (method, path) {
        ("GET", "/") => ok("home"),
        ("GET", "/health") => ok("ok"),
        ("POST", "/orders") => Response { status: 201, body: "order created" },
        ("GET", _) => {
            if parts.len() == 3 && parts[1] == "users" {
                user_page(parts[2])
            } else {
                not_found(path)
            }
        },
        _ => Response { status: 405, body: "method not allowed" },
    }
}

    let calls = [("GET", "/"), ("GET", "/health"), ("POST", "/orders"), ("GET", "/users/42"), ("GET", "/users/abc"), ("GET", "/nowhere"), ("DELETE", "/")]

// idiom: a struct holds balance and a Vec of entries; every change goes through one method that validates and records it
struct Entry {
    kind: string,
    amount: int,
    balance_after: int,
}

struct Account {
    owner: string,
    balance: int,
    history: Vec<Entry>,
}

impl Account {
    fn deposit(self, amount: int) -> Result<int, string> {
        if amount <= 0 {
            return Err("deposit must be positive")
        }
        self.balance += amount
        self.history.push(Entry { kind: "deposit", amount: amount, balance_after: self.balance })
        Ok(self.balance)
    }

    fn withdraw(self, amount: int) -> Result<int, string> {
        if amount <= 0 {
            return Err("withdrawal must be positive")
        }
        if amount > self.balance {
            return Err("insufficient funds: balance ${self.balance}, requested ${amount}")
        }
        self.balance -= amount
        self.history.push(Entry { kind: "withdrawal", amount: amount, balance_after: self.balance })
        Ok(self.balance)
    }

    fn statement(self) {
        println("statement for ${self.owner}")
        for e in self.history {
            println("  ${e.kind:<10} ${e.amount:>6} -> ${e.balance_after:>6}")
        }
    }
}

// golden: double-entry bookkeeping
// idiom: every transaction is a list of postings that must sum to zero; a HashMap keeps the total per account
use std::collections::HashMap

struct Posting {
    account: string,
    amount: int,
}

fn balanced(postings: Vec<Posting>) -> bool {
    let mut sum = 0
    for p in postings {
        sum += p.amount
    }
    sum == 0
}

fn post(books: HashMap<string, int>, postings: Vec<Posting>) -> Result<HashMap<string, int>, string> {
    if !balanced(postings) {
        return Err("postings do not sum to zero")
    }
    let mut next = books
    for p in postings {
        let current = next.get(p.account) ?? 0
        next.insert(p.account, current + p.amount)
    }
    Ok(next)
}

fn main() {
    let mut books: HashMap<string, int> = HashMap::new()
    let entries = [
        [Posting { account: "cash", amount: 1000 }, Posting { account: "equity", amount: -1000 }],
        [Posting { account: "inventory", amount: 400 }, Posting { account: "cash", amount: -400 }],
        [Posting { account: "cash", amount: 150 }, Posting { account: "sales", amount: -100 }],
        [Posting { account: "cash", amount: 90 }, Posting { account: "sales", amount: -90 }],
    ]
    for postings in entries {
        match post(books, postings) {
            Ok(next) => books = next,
            Err(e) => println("rejected: ${e}"),
        }
    }
    let mut names: Vec<string> = books.keys().collect()
    names.sort()
    let mut total = 0
    for name in names {
        println("${name:<10} ${books[name]:>6}")
        total += books[name]
    }
    println("books balance: ${total == 0}")
}

=== parse_calculator.wj ===
// golden: recursive descent calculator
// idiom: a tokenizer produces a Vec of tokens; a parser struct walks them with an index and one method per grammar rule
enum Token {
    Num(int),
    Op(char),
}

fn tokenize(text: string) -> Result<Vec<Token>, string> {
    let mut tokens: Vec<Token> = []
    let cs: Vec<char> = text.chars().collect()
    let mut i = 0
    while i < cs.len() {
        let c = cs[i]
        if c == ' ' {
            i += 1
        } else if c.is_ascii_digit() {
            let mut n = 0
            while i < cs.len() && cs[i].is_ascii_digit() {
                n = n * 10 + (cs[i] as int - '0' as int)
                i += 1
            }
            tokens.push(Token::Num(n))
        } else if "+-*/()".contains("${c}") {
            tokens.push(Token::Op(c))
            i += 1
        } else {
            return Err("unexpected character '${c}'")
        }
    }
    Ok(tokens)
}

struct Parser {
    tokens: Vec<Token>,
    pos: int,
}

impl Parser {
    fn peek_op(self) -> char {
        if self.pos < self.tokens.len() {
            match self.tokens[self.pos] {
                Token::Op(c) => c,
                Token::Num(_) => ' ',
            }
        } else {
            ' '
        }
    }

    fn expr(self) -> Result<int, string> {
        let mut value = self.term()?
        while self.peek_op() == '+' || self.peek_op() == '-' {
            let op = self.peek_op()
            self.pos += 1
            let rhs = self.term()?
            value = if op == '+' { value + rhs } else { value - rhs }
        }
        Ok(value)
    }

    fn term(self) -> Result<int, string> {
        let mut value = self.factor()?
        while self.peek_op() == '*' || self.peek_op() == '/' {
            let op = self.peek_op()
            self.pos += 1
            let rhs = self.factor()?
            if op == '/' && rhs == 0 {
                return Err("division by zero")
            }
            value = if op == '*' { value * rhs } else { value / rhs }
        }
        Ok(value)
    }

    fn factor(self) -> Result<int, string> {
        if self.pos >= self.tokens.len() {
            return Err("unexpected end of input")
        }
        match self.tokens[self.pos] {
            Token::Num(n) => {
                self.pos += 1
                Ok(n)
            },
            Token::Op('(') => {
                self.pos += 1
                let v = self.expr()?
                if self.peek_op() != ')' {
                    return Err("missing )")
                }
                self.pos += 1
                Ok(v)
            },
            Token::Op(c) => Err("unexpected '${c}'"),
        }
    }
}

fn calc(text: string) -> Result<int, string> {
    let mut p = Parser { tokens: tokenize(text)?, pos: 0 }
    let v = p.expr()?
    if p.pos < p.tokens.len() {
        return Err("trailing input")
    }
    Ok(v)
}

fn main() {
    for text in ["1 + 2 * 3", "(1 + 2) * 3", "10 / (5 - 5)", "2 * (3 + 4) - 5", "2 $ 3", "(1 + 2", "4 +"] {
        match calc(text) {
            Ok(v) => println("${text} = ${v}"),
            Err(e) => println("${text} -> error: ${e}"),
        }
    }
}
--- output ---

(1 + 2) * 3 = 9
10 / (5 - 5) -> error: division by zero
2 * (3 + 4) - 5 = 9
2 $ 3 -> error: unexpected character '$'

// golden: INI-style config parser
// idiom: one pass over lines with a "current section" variable; entries are structs; strings.trim returns an owned string
use std::strings

struct Entry {
    section: string,
    key: string,
    value: string,
}

fn parse(text: string) -> Result<Vec<Entry>, string> {
    let mut entries: Vec<Entry> = []
    let mut section = "global"
    let mut number = 0
    for raw in text.lines() {
        number += 1
        let line = strings.trim(raw)
        if line.len() == 0 || line.starts_with("#") {
            continue
        }
        if line.starts_with("[") {
            if !line.ends_with("]") {
                return Err("line ${number}: unterminated section header")
            }
            section = line[1..line.len() - 1]
        } else if line.contains("=") {
            let kv: Vec<string> = line.split("=").collect()
            entries.push(Entry { section: section, key: strings.trim(kv[0]), value: strings.trim(kv[1]) })
        } else {
            return Err("line ${number}: expected key = value")
        }
    }
    Ok(entries)
}

fn get(entries: Vec<Entry>, section: string, key: string) -> Option<string> {
    for e in entries {
        if e.section == section && e.key == key {
            return Some(e.value)
        }
    }
    None
}

// golden: Roman numerals both ways
// idiom: a table of (value, symbol) pairs drives both directions; greedy subtraction and a left-to-right scan
fn table() -> Vec<(int, string)> {
    [(1000, "M"), (900, "CM"), (500, "D"), (400, "CD"), (100, "C"), (90, "XC"), (50, "L"), (40, "XL"), (10, "X"), (9, "IX"), (5, "V"), (4, "IV"), (1, "I")]
}

fn to_roman(n: int) -> string {
    let mut left = n
    let mut out = ""
    for (value, symbol) in table() {
        while left >= value {
            out += symbol
            left -= value
        }
    }
    out
}

fn symbol_value(c: char) -> int {
    match c {
        'I' => 1,
        'V' => 5,
        'X' => 10,
        'L' => 50,
        'C' => 100,
        'D' => 500,
        'M' => 1000,
        _ => 0,
    }
}

fn from_roman(text: string) -> int {
    let cs: Vec<char> = text.chars().collect()
    let mut total = 0
    let mut i = 0
    while i < cs.len() {
        let v = symbol_value(cs[i])
        if i + 1 < cs.len() && v < symbol_value(cs[i + 1]) {
            total -= v
        } else {
            total += v
        }
        i += 1
    }
    total
}

fn main() {
    for n in [1, 4, 9, 14, 40, 90, 400, 1994, 2024, 3999] {
        let r = to_roman(n)
        println("${n} -> ${r} -> ${from_roman(r)}")
    }
}

// golden: tokenizer for a tiny expression language
// idiom: classify characters with is_alphabetic / is_ascii_digit; a Vec of token structs carries kind and text
struct Token {
    kind: string,
    text: string,
}

fn tokenize(source: string) -> Vec<Token> {
    let cs: Vec<char> = source.chars().collect()
    let mut tokens: Vec<Token> = []
    let mut i = 0
    while i < cs.len() {
        let c = cs[i]
        if c.is_whitespace() {
            i += 1
        } else if c.is_alphabetic() || c == '_' {
            let mut word = ""
            while i < cs.len() && (cs[i].is_alphanumeric() || cs[i] == '_') {
                word += "${cs[i]}"
                i += 1
            }
            let kind = if word == "let" || word == "if" || word == "else" { "keyword" } else { "ident" }
            tokens.push(Token { kind: kind, text: word })
        } else if c.is_ascii_digit() {

// golden: Game of Life
// idiom: Vec<Vec<bool>> grid; neighbor counting with bounds checks on int coordinates; one pure step function
fn alive_neighbors(grid: Vec<Vec<bool>>, row: int, col: int) -> int {
    let mut count = 0
    for dr in [-1, 0, 1] {
        for dc in [-1, 0, 1] {
            if dr == 0 && dc == 0 {
                continue
            }
            let r = row + dr
            let c = col + dc
            if r >= 0 && r < grid.len() as int && c >= 0 && c < grid[0].len() as int && grid[r][c] {
                count += 1
            }
        }
    }
    count
}

fn step(grid: Vec<Vec<bool>>) -> Vec<Vec<bool>> {
    let mut next: Vec<Vec<bool>> = []
    for r in 0..grid.len() {
        let mut row: Vec<bool> = []
        for c in 0..grid[0].len() {
            let n = alive_neighbors(grid, r, c)
            row.push(if grid[r][c] { n == 2 || n == 3 } else { n == 3 })
        }
        next.push(row)
    }
    next
}

fn show(grid: Vec<Vec<bool>>) {
    for row in grid {
        let cells: Vec<string> = row.map(|alive| if alive { "#" } else { "." }).collect()
        println(cells.join(""))
    }
}

// golden: seeded random walk
// idiom: a small struct owns the generator state; methods mutate it; results are fully deterministic
struct Walker {
    seed: int,
    x: int,
    y: int,
    steps: int,
}

impl Walker {
    fn roll(self) -> int {
        self.seed = (self.seed * 1103515245 + 12345) % 2147483648
        self.seed / 65536 % 4
    }

    fn step(self) {
        let dir = self.roll()
        match dir {
            0 => { self.x += 1 },
            1 => { self.x -= 1 },
            2 => { self.y += 1 },
            _ => { self.y -= 1 },
        }
        self.steps += 1
    }

    fn distance(self) -> int {
        let dx = if self.x < 0 { 0 - self.x } else { self.x }
        let dy = if self.y < 0 { 0 - self.y } else { self.y }
        dx + dy
    }
}

fn main() {
    let mut w = Walker { seed: 7, x: 0, y: 0, steps: 0 }
    let mut farthest = 0
    for i in 0..50 {
        w.step()
        if w.distance() > farthest {
            farthest = w.distance()
        }
        if (i + 1) % 10 == 0 {
            println("after ${w.steps} steps at (${w.x}, ${w.y}), distance ${w.distance()}")
        }
    }
    println("farthest distance ${farthest}")
}

// idiom: list the legal transitions in one match; illegal ones return Err with both state names
enum Status {
    Draft,
    Paid,
    Shipped,
    Delivered,
    Cancelled,
}

fn name(s: Status) -> string {
    match s {
        Status::Draft => "draft",
        Status::Paid => "paid",
        Status::Shipped => "shipped",
        Status::Delivered => "delivered",
        Status::Cancelled => "cancelled",
    }
}

fn advance(from: Status, to: Status) -> Result<Status, string> {
    match (from, to) {
        (Status::Draft, Status::Paid) => Ok(Status::Paid),
        (Status::Paid, Status::Shipped) => Ok(Status::Shipped),
        (Status::Shipped, Status::Delivered) => Ok(Status::Delivered),
        (Status::Draft, Status::Cancelled) => Ok(Status::Cancelled),
        (Status::Paid, Status::Cancelled) => Ok(Status::Cancelled),
        (a, b) => Err("cannot go from ${name(a)} to ${name(b)}"),
    }
}

fn run(steps: Vec<Status>) {
    let mut current = Status::Draft
    for target in steps {
        match advance(current, target) {
            Ok(next) => {
                println("${name(current)} -> ${name(next)}")
                current = next
            },
            Err(e) => {
                println("stop: ${e}")
                return
            },
        }
    }
    println("finished in ${name(current)}")
}

fn main() {
    run([Status::Paid, Status::Shipped, Status::Delivered])
    run([Status::Paid, Status::Cancelled])
    run([Status::Shipped])
    run([Status::Cancelled, Status::Paid])
}

struct Machine {
    credit: int,
    stock: int,
    price: int,
    sales: int,
}

    fn insert(self, coins: int) {
        self.credit += coins
    }

    fn press(self) -> Result<string, string> {
        if self.stock == 0 {
            return Err("sold out")
        }
        if self.credit < self.price {
            return Err("need ${self.price - self.credit} more")

        self.credit -= self.price

        self.sales += self.price

        let back = self.credit
        self.credit = 0
        back
    }
}

        Err(e) => println("press: refused (${e})"),
    }
}

// golden: Caesar cipher
// idiom: chars() walks a string; `as int` turns a char into a number, `as u8` then `as char` turns a number back
fn shift_char(c: char, key: int) -> char {
    if c >= 'a' && c <= 'z' {
        let offset = (c as int - 'a' as int + key + 26) % 26
        return (('a' as int + offset) as u8) as char
    }
    if c >= 'A' && c <= 'Z' {
        let offset = (c as int - 'A' as int + key + 26) % 26
        return (('A' as int + offset) as u8) as char
    }
    c
}

fn encrypt(text: string, key: int) -> string {
    let mut out = ""
    for c in text.chars() {
        out += "${shift_char(c, key)}"
    }
    out
}

fn wrap(text: string, width: int) -> Vec<string> {
    let mut lines: Vec<string> = []
    let mut line = ""
    for word in strings.split_whitespace(text) {
        if line.len() == 0 {
            line = word
        } else if line.len() as int + 1 + word.len() as int <= width {
            line = "${line} ${word}"

        }

// golden: run-length encoding
// idiom: walk a Vec<char> with a while loop and an index; build output by appending interpolated pieces
fn encode(text: string) -> string {
    let cs: Vec<char> = text.chars().collect()
    let mut out = ""
    let mut i = 0
    while i < cs.len() {
        let c = cs[i]
        let mut run = 1
        while i + run < cs.len() && cs[i + run] == c {
            run += 1
        }
        out += "${run}${c}"
        i += run
    }
    out
}

fn decode(text: string) -> string {
    let mut out = ""
    let mut count = 0
    for c in text.chars() {
        if c.is_ascii_digit() {
            count = count * 10 + (c as int - '0' as int)
        } else {
            for _ in 0..count {
                out += "${c}"
            }
            count = 0
        }
    }
    out
}

// golden: enum with data and match
// idiom: variants carry data; match is exhaustive; arms are expressions; struct variants name their fields
enum Shape {
    Circle(float),
    Rect(float, float),
    Triangle { base: float, height: float },
}

fn area(s: Shape) -> float {
    match s {
        Shape::Circle(r) => 3.14159 * r * r,
        Shape::Rect(w, h) => w * h,
        Shape::Triangle { base, height } => 0.5 * base * height,
    }
}

fn name(s: Shape) -> string {
    match s {
        Shape::Circle(_) => "circle",
        Shape::Rect(w, h) => if w == h { "square" } else { "rectangle" },
        Shape::Triangle { .. } => "triangle",
    }
}

fn describe(s: Shape) -> string {
    "${name(s)} with area ${area(s):.2}"
}

fn main() {
    let shapes = [
        Shape::Circle(2.0),
        Shape::Rect(3.0, 4.0),
        Shape::Rect(5.0, 5.0),
        Shape::Triangle { base: 6.0, height: 2.5 },
    ]
    let mut total = 0.0
    for s in shapes {
        println(describe(s))
        total += area(s)
    }
    println("total area ${total:.2}")
}

rectangle with area 12.00

=== types_generic_container.wj ===
// golden: generic container
// idiom: struct Stack<T> with generic methods; the compiler infers T from usage, no turbofish needed
struct Stack<T> {
    items: Vec<T>,
}

impl<T> Stack<T> {
    fn new() -> Stack<T> {
        Stack { items: [] }
    }

    fn push(self, item: T) {
        self.items.push(item)
    }

    fn pop(self) -> Option<T> {
        self.items.pop()
    }

    fn peek(self) -> Option<T> {
        if self.items.len() == 0 {
            return None
        }
        Some(self.items[self.items.len() - 1])
    }

    fn size(self) -> int {
        self.items.len() as int
    }

    fn is_empty(self) -> bool {
        self.items.len() == 0
    }
}

fn main() {
    let mut numbers: Stack<int> = Stack::new()
    numbers.push(10)
    numbers.push(20)
    numbers.push(30)
    println("size ${numbers.size()}, top ${numbers.peek() ?? -1}")
    println("popped ${numbers.pop() ?? -1}")
    println("size ${numbers.size()}, empty ${numbers.is_empty()}")
    let mut words: Stack<string> = Stack::new()
    words.push("alpha")
    words.push("beta")
    println("words top ${words.peek() ?? "none"}")
    let mut empty: Stack<string> = Stack::new()
    println("empty pop ${empty.pop() ?? "nothing"}")
}
--- output ---

popped 30
size 2, empty false
words top beta
empty pop nothing

// golden: small structs as units
// idiom: wrap a number in a struct so meters and seconds cannot mix; derive-free equality and ordering by field
struct Meters {
    value: float,
}

struct Seconds {
    value: float,
}

struct Speed {
    mps: float,
}

fn speed(distance: Meters, time: Seconds) -> Speed {
    Speed { mps: distance.value / time.value }
}

fn kmh(s: Speed) -> float {
    s.mps * 3.6
}

fn faster(a: Speed, b: Speed) -> Speed {
    if a.mps >= b.mps { a } else { b }
}

fn main() {
    let sprint = speed(Meters { value: 100.0 }, Seconds { value: 9.58 })
    let marathon = speed(Meters { value: 42195.0 }, Seconds { value: 7299.0 })
    println("sprint ${sprint.mps:.2} m/s = ${kmh(sprint):.1} km/h")
    println("marathon ${marathon.mps:.2} m/s = ${kmh(marathon):.1} km/h")
    let winner = faster(sprint, marathon)
    println("faster: ${winner.mps:.2} m/s")
    let same = Meters { value: 5.0 } == Meters { value: 5.0 }
    println("equal values compare equal: ${same}")
}

=== types_option_chaining.wj ===
// golden: Option handling
// idiom: match, if let, `??` default, map and and_then chain Option values without unwrap
struct User {
    name: string,
    email: Option<string>,
    age: Option<int>,
}

fn domain(email: string) -> Option<string> {
    let parts: Vec<string> = email.split("@").collect()
    if parts.len() == 2 { Some(parts[1]) } else { None }
}

fn email_domain(u: User) -> Option<string> {
    u.email.and_then(|e| domain(e))
}

fn age_group(u: User) -> string {
    match u.age {
        Some(a) if a < 18 => "minor",
        Some(a) if a < 65 => "adult",
        Some(_) => "senior",
        None => "unknown",
    }
}

fn main() {
    let users = [
        User { name: "ann", email: Some("ann@example.com"), age: Some(34) },
        User { name: "bob", email: None, age: Some(15) },
        User { name: "cy", email: Some("not-an-email"), age: None },
        User { name: "di", email: Some("di@wj.dev"), age: Some(70) },
    ]
    for u in users {
        let who = u.name
        let shown = email_domain(u) ?? "no domain"
        println("${who}: ${shown}, ${age_group(u)}")
        if let Some(a) = u.age {
            println("  next year ${a + 1}")
        }
    }
    let doubled = Some(21).map(|x| x * 2)
    println("doubled ${doubled ?? 0}")
}
--- output ---

  next year 35
bob: no domain, minor
  next year 16
cy: no domain, unknown
di: wj.dev, senior
  next year 71

=== types_struct_methods.wj ===
// golden: struct with methods
// idiom: methods take plain self; the compiler infers read, mutate or consume; mutation needs a `let mut` owner
struct Counter {
    name: string,
    count: int,
    step: int,
}

impl Counter {
    fn new(name: string, step: int) -> Counter {
        Counter { name: name, count: 0, step: step }
    }

    fn tick(self) {
        self.count += self.step
    }

    fn reset(self) {
        self.count = 0
    }

    fn describe(self) -> string {
        "${self.name}=${self.count}"
    }

    fn is_past(self, limit: int) -> bool {
        self.count > limit
    }
}

fn main() {
    let mut a = Counter::new("a", 1)
    let mut b = Counter::new("b", 5)
    for _ in 0..3 {
        a.tick()
        b.tick()
    }
    println(a.describe())
    println(b.describe())
    println("b past 10: ${b.is_past(10)}")
    b.reset()
    println("after reset: ${b.describe()}")
    println("a past 10: ${a.is_past(10)}")
}
--- output ---
a=3
b=15
b past 10: true
after reset: b=0
a past 10: false

=== types_trait_shapes.wj ===
// golden: traits and implementations
// idiom: a trait lists method signatures with plain self; each type implements it; generics accept any implementor
trait Describe {
    fn describe(&self) -> string
}

struct Dog {
    name: string,
    age: int,
}

struct Robot {
    model: string,
    battery: int,
}

impl Describe for Dog {
    fn describe(self) -> string {
        "${self.name} the dog, ${self.age} years old"
    }
}

impl Describe for Robot {
    fn describe(self) -> string {
        "robot ${self.model} at ${self.battery}% battery"
    }
}

fn announce<T: Describe>(item: T) {
    println("-> ${item.describe()}")
}

fn longest<T: Describe>(a: T, b: T) -> string {
    let da = a.describe()
    let db = b.describe()
    if da.len() >= db.len() { da } else { db }
}

fn main() {
    let rex = Dog { name: "Rex", age: 4 }
    let bolt = Dog { name: "Bolt", age: 11 }
    let r2 = Robot { model: "R2", battery: 80 }
    announce(rex)
    announce(r2)
    println(longest(rex, bolt))
    println("done")
}
--- output ---
-> Rex the dog, 4 years old
-> robot R2 at 80% battery
Bolt the dog, 11 years old
done
